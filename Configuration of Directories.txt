1. 
mkdir AcmeCA 
cd AcmeCA 
mkdir -p ca/{root-ca,sub-ca,server}/{certs,crl,newcerts,private,csr} 
2. tree ca

3. chmod -v 700 ca/{root-ca,sub-ca,server}/private 

4. touch ca/{root-ca,sub-ca,server}/index 
5. 
openssl rand -hex 16 > ca/root-ca/serial 
openssl rand -hex 16 > ca/sub-ca/serial 
tree ca 

6. 
cd ca 
openssl genrsa -aes256 -out root-ca/private/ca.key 4096 
--They will ask for a password.


7. gedit root-ca/root-ca.conf 

8. 
cd root-ca 
tree 

9.
openssl req -config root-ca.conf -key private/ca.key -new -x509 -days 7305 -sha256 -extensions v3_ca -out certs/ca.crt
--It will ask for a private key password.

10. openssl x509 -noout -text -in certs/ca.crt 

11. cd ../ 
12. openssl genrsa -aes256 -out sub-ca/private/sub-ca.key 4096 
--give any password:  

13.
cd sub-ca 
gedit sub-ca.conf 

14.
openssl req -config sub-ca.conf -new -sha256 -key private/sub-ca.key -out csr/sub-ca.csr 

15.
cd ../ 
cd root-ca 
openssl ca -config root-ca.conf -extensions v3_intermediate_ca -days 3650 -notext in ../sub-ca/csr/sub-ca.csr -out ../sub-ca/certs/sub-ca.crt 

16.
tree 
cat index 

17. openssl x509 -noout -text -in ../sub-ca/certs/sub-ca.crt 

18. 
cd ../ 
openssl genrsa -out server/private/server.key 2048 

19.
cd server 
openssl req -key private/server.key -new -sha256 -out csr/server.csr

cd ../sub-ca/
