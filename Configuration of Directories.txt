3. chmod -v 700 ca/{root-ca,sub-ca,server}/private 

4. touch ca/{root-ca,sub-ca,server}/index

openssl rand -hex 16 > ca/root-ca/serial

cd ca

cd root-ca

10. openssl x509 -noout -text -in certs/ca.crt
