## Install all the pre-requisite tools for project one by one.
1) sudo apt update 
2) sudo apt install snort  
3) sudo apt install bind9 
4) sudo apt install apache2 
5) sudo apt install ufw  
6) sudo apt install net-tools 
7) sudo apt install python3 
8) sudo apt install python3 python3-pip 
9) pip install Flask 
10) pip install request

## Verify 
snort --version
named -v
apache2 -v
ufw --version
ifconfig --version
python3 --version
pip3 --version
python3 -c "import flask; print(flask.__version__)"
